cmake_minimum_required(VERSION 3.24)
project(xgb_consumer LANGUAGES C)

set(XGB_CONSUMER_MODE installed CACHE STRING "Package consumption contract")
set(XGB_CONSUMER_SOURCE_DIR "" CACHE PATH "Explicit component source checkout")
option(XGB_CONSUMER_CHECK_CXX "Build an independent C++ header/link consumer" OFF)

if(XGB_CONSUMER_MODE MATCHES "_(preexisting|bad_version|bad_abi|no_identity)$")
    include("${xgen_bytes_DIR}/xgenBytesTargets.cmake")
    if(XGB_CONSUMER_MODE MATCHES "_bad_version$")
        set_property(TARGET xgb::bytes PROPERTY XGB_VERSION 9.0.0)
    elseif(XGB_CONSUMER_MODE MATCHES "_bad_abi$")
        set_property(TARGET xgb::bytes PROPERTY XGB_ABI_VERSION 9)
    elseif(XGB_CONSUMER_MODE MATCHES "_no_identity$")
        set_property(TARGET xgb::bytes PROPERTY XGB_VERSION)
        set_property(TARGET xgb::bytes PROPERTY XGB_ABI_VERSION)
    endif()
endif()

if(XGB_CONSUMER_MODE MATCHES "^source")
    set(XGB_BUILD_TESTS OFF CACHE BOOL "Consumer disables upstream tests" FORCE)
    set(XGB_ENABLE_COVERAGE OFF CACHE BOOL "Consumer uses production library" FORCE)
    add_subdirectory("${XGB_CONSUMER_SOURCE_DIR}" dependency)
elseif(XGB_CONSUMER_MODE STREQUAL "installed_optional")
    find_package(xgen_bytes 0.1.0 EXACT CONFIG REQUIRED
        COMPONENTS bytes OPTIONAL_COMPONENTS missing)
    if(xgen_bytes_missing_FOUND)
        message(FATAL_ERROR "Unknown optional component was accepted")
    endif()
elseif(XGB_CONSUMER_MODE STREQUAL "installed_unknown")
    find_package(xgen_bytes 0.1.0 EXACT CONFIG REQUIRED COMPONENTS missing)
elseif(XGB_CONSUMER_MODE STREQUAL "installed_wrong_version")
    find_package(xgen_bytes 0.2.0 EXACT CONFIG REQUIRED COMPONENTS bytes)
elseif(XGB_CONSUMER_MODE STREQUAL "installed_default")
    find_package(xgen_bytes 0.1.0 EXACT CONFIG REQUIRED)
else()
    find_package(xgen_bytes 0.1.0 EXACT CONFIG REQUIRED COMPONENTS bytes)
endif()

if(XGB_CONSUMER_MODE MATCHES "_(bad_version|bad_abi|no_identity)$")
    message(STATUS "Incompatible preexisting target was incorrectly accepted")
    return()
endif()
if(NOT TARGET xgb::bytes)
    message(FATAL_ERROR "The public bytes target is missing")
endif()
get_target_property(version xgb::bytes XGB_VERSION)
get_target_property(abi xgb::bytes XGB_ABI_VERSION)
if(NOT version VERSION_EQUAL "0.1.0" OR NOT abi STREQUAL "1")
    message(FATAL_ERROR "Public version/ABI properties were not preserved")
endif()
get_target_property(dependencies xgb::bytes INTERFACE_LINK_LIBRARIES)
if(dependencies OR TARGET GTest::gtest OR TARGET GTest::gtest_main OR TARGET xgc::base)
    message(FATAL_ERROR "Production bytes must not import library dependencies")
endif()

add_executable(consumer_c consumer.c)
set_target_properties(consumer_c PROPERTIES
    C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(consumer_c PRIVATE xgb::bytes)
enable_testing()
add_test(NAME consumer_c_runs COMMAND consumer_c)
if(XGB_CONSUMER_CHECK_CXX)
    enable_language(CXX)
    add_executable(consumer_cpp consumer.cpp)
    set_target_properties(consumer_cpp PROPERTIES
        CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    target_link_libraries(consumer_cpp PRIVATE xgb::bytes)
    add_test(NAME consumer_cpp_runs COMMAND consumer_cpp)
endif()
